可视化生成器

一键生成高质量科研图表，支持多种可视化类型和自定义样式。

import matplotlib.pyplot as plt
import seaborn as sns

def generate_publication_plot(data, plot_type, save_path):
    # 设置科研出版物风格
    plt.style.use('seaborn-whitegrid')
    sns.set_context("paper")
    
    fig, ax = plt.subplots(figsize=(8, 6))
    
    if plot_type == "scatter":
        sns.scatterplot(data=data, x="x_var", y="y_var", 
                       hue="category", size="value", ax=ax)
    elif plot_type == "line":
        sns.lineplot(data=data, x="x_var", y="y_var", 
                    hue="category", style="category", ax=ax)
                    
    # 设置标题和标签
    ax.set_title("Research Results", fontsize=16)
    ax.set_xlabel("Variable X", fontsize=12)
    ax.set_ylabel("Variable Y", fontsize=12)
    
    # 保存图表
    plt.tight_layout()
    plt.savefig(save_path, dpi=300)

    # 评估模型
    results = model.evaluate(X_test, y_test)
    
    # 记录指标
    mlflow.log_metrics({
        "test_loss": results[0],
        "test_accuracy": results[1]
    })
    
    # 保存模型
    model.save(os.path.join(save_dir, "model.h5"))
    mlflow.end_run()

实验结果分析器

自动分析实验结果，执行统计测试并生成报告。

import pandas as pd
import numpy as np
import scipy.stats as stats
from reportlab.lib import colors
from reportlab.lib.pagesizes import letter
from reportlab.platypus import SimpleDocTemplate, Table, TableStyle

def analyze_experiment_results(results_file, output_pdf):
    # 读取实验结果
    df = pd.read_csv(results_file)
    
    # 基本统计分析
    stats_summary = df.groupby('treatment').agg({
        'response': ['mean', 'std', 'min', 'max', 'count']
    })
    
    # 执行统计测试
    groups = [df[df['treatment'] == t]['response'] for t in df['treatment'].unique()]
    f_stat, p_value = stats.f_oneway(*groups)
    
    # 多重比较
    tukey_results = stats.tukey_hsd(*groups)
    
    # 生成PDF报告
    doc = SimpleDocTemplate(output_pdf, pagesize=letter)
    elements = []
    
    # 添加表格
    data = [['Treatment', 'Mean', 'Std Dev', 'Min', 'Max', 'Count']]
    for idx, row in stats_summary.iterrows():
        data.append([idx] + list(row['response']))
    
    # 添加ANOVA结果
    data.append(['ANOVA F-statistic', f_stat, '', '', '', ''])
    data.append(['ANOVA p-value', p_value, '', '', '', ''])
    
    # 创建表格
    t = Table(data)
    t.setStyle(TableStyle([
        ('BACKGROUND', (0, 0), (-1, 0), colors.grey),
        ('TEXTCOLOR', (0, 0), (-1, 0), colors.whitesmoke),
        ('ALIGN', (0, 0), (-1, -1), 'CENTER'),
        ('FONTNAME', (0, 0), (-1, 0), 'Helvetica-Bold'),
        ('BOTTOMPADDING', (0, 0), (-1, 0), 12),
        ('GRID', (0, 0), (-1, -1), 1, colors.black)
    ]))
    
    elements.append(t)
    doc.build(elements)
